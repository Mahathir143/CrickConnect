2. Sample Automated Test Implementation (Vitest / Jest)

import { describe, it, expect } from 'vitest';
import { processBallEvent, rollbackLastBall } from '../src/core/cricketRules';

describe('Cricket Scoring & ICC Rulebook Verification', () => {
  it('TC-05: correctly computes No-Ball + 4 off bat and activates Free Hit', () => {
    const initialState = createFreshMatchState();
    const event = {
      batsmanRuns: 4,
      extraType: 'NO_BALL',
      extraRuns: 1,
      isLegalDelivery: false,
    };

    const nextState = processBallEvent(initialState, event);

    expect(nextState.battingTeam.score).toBe(5);
    expect(nextState.striker.runs).toBe(4);
    expect(nextState.striker.fours).toBe(1);
    expect(nextState.isFreeHit).toBe(true);
    expect(nextState.oversCompleted).toBe(0);
    expect(nextState.ballsThisOver).toBe(0); // Did not consume legal ball
  });

  it('TC-09: verifies complete state restoration on Undo', () => {
    let state = createFreshMatchState();
    const originalStrikerId = state.striker.id;

    // Ball 1: Single taken
    state = processBallEvent(state, { batsmanRuns: 1, extraRuns: 0, isLegalDelivery: true });
    expect(state.striker.id).not.toBe(originalStrikerId);

    // Ball 2: Boundary four
    state = processBallEvent(state, { batsmanRuns: 4, extraRuns: 0, isLegalDelivery: true });
    expect(state.battingTeam.score).toBe(5);

    // Undo Ball 2
    state = rollbackLastBall(state);
    expect(state.battingTeam.score).toBe(1);
    expect(state.ballsThisOver).toBe(1);

    // Undo Ball 1
    state = rollbackLastBall(state);
    expect(state.battingTeam.score).toBe(0);
    expect(state.striker.id).toBe(originalStrikerId);
  });
});
